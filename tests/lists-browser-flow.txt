async (page) => {
  page.setDefaultTimeout(10000);
  // Slow local directory responses expose late-render races during file selection.
  await page.route('**/api/wordpress/lists/directory', async route => { await new Promise(r => setTimeout(r, 150)); await route.continue(); });
  const errors = [];
  page.on('pageerror', e => errors.push(e.message));
  const cdp = await page.context().newCDPSession(page);
  await cdp.send('WebAuthn.disable');
  await cdp.send('WebAuthn.enable');
  await cdp.send('WebAuthn.addVirtualAuthenticator', {options: {protocol:'ctap2',transport:'internal',hasResidentKey:true,hasUserVerification:true,isUserVerified:true,automaticPresenceSimulation:true}});
  await page.goto('about:blank');
  await page.goto('http://localhost:4326/?ui=lists#setup=browser-test-invite');
  await page.getByRole('button', {name:'Create your passkey',exact:true}).click();
  await page.getByRole('button', {name:'Generate backup codes',exact:true}).click();
  await page.getByLabel('I have saved my backup codes somewhere safe.').check();
  await page.getByRole('button', {name:'Confirm saved',exact:true}).click();
  await page.getByRole('button', {name:'Manage lists',exact:true}).click();
  await page.getByRole('heading', {name:'Lists.',exact:true}).waitFor();
  await page.getByRole('button', {name:'Refresh follows',exact:true}).click();
  await page.waitForFunction(async () => (await (await fetch('/api/wordpress/lists/directory')).json()).accounts.length === 2);
  await page.getByRole('button', {name:'Refresh follows',exact:true}).waitFor();
  const csv = 'Science,sample@example.test\nFriends,sample@example.test\nFriends,old@example.test\nScience,another@example.test';
  const upload = async () => {
    await page.getByLabel('Mastodon lists.csv').setInputFiles({name:'lists.csv',mimeType:'text/csv',buffer:Buffer.from(csv)});
    await page.getByRole('button', {name:'Preview import',exact:true}).click();
    await page.getByRole('heading', {name:'Preview Mastodon list import',exact:true}).waitFor();
    if (!await page.getByText('2 lists · 4 memberships · 3 accounts',{exact:true}).isVisible()) throw Error('Import summary');
    await page.getByRole('button', {name:'Import lists',exact:true}).click();
    await page.waitForFunction(() => !document.querySelector('#modal').open);
  };
  await upload();
  await page.getByRole('button', {name:'Friends 1 matched · 1 unmatched',exact:true}).click();
  await page.getByRole('heading', {name:'Friends',exact:true}).waitFor();
  if (page.viewportSize().width <= 1050 && !await page.locator('#list-detail').evaluate(e => e.getBoundingClientRect().top < innerHeight)) throw Error('Mobile editor not brought into view');
  await page.getByLabel('List name',{exact:true}).fill('Close circle');
  await page.getByRole('button', {name:'Rename',exact:true}).click();
  await page.getByRole('heading', {name:'Close circle',exact:true}).waitFor();
  await page.getByRole('button', {name:'Read this list',exact:true}).click();
  await page.waitForFunction(() => document.querySelectorAll('.board .post').length === 1);
  if (await page.locator('.board .author-info').first().textContent().then(t => !t.includes('Sample author'))) throw Error('Feed list filter');
  const id = await page.locator('#feed-list').inputValue();
  await page.getByRole('button',{name:'People',exact:true}).first().click();
  await page.waitForFunction(() => document.querySelectorAll('.person-row').length === 1);
  await page.locator('.person-row').getByRole('button',{name:'Lists',exact:true}).click();
  if (!await page.getByRole('checkbox',{name:'Science',exact:true}).isChecked() || !await page.getByRole('checkbox',{name:'Close circle',exact:true}).isChecked()) throw Error('Multiple list membership');
  await page.getByRole('checkbox',{name:'Close circle',exact:true}).uncheck();
  await page.getByRole('button',{name:'Save list choices',exact:true}).click();
  await page.waitForFunction(() => !document.querySelector('#modal').open);
  await page.getByRole('button',{name:'Manage lists',exact:true}).click();
  await upload();
  await page.getByRole('button',{name:'Close circle 0 matched · 1 unmatched',exact:true}).click();
  await page.getByText('Unmatched accounts · 1',{exact:true}).click();
  await page.getByRole('button',{name:'Match account',exact:true}).click();
  await page.getByLabel('Current followed account').selectOption('another@example.test');
  await page.getByRole('button',{name:'Save match',exact:true}).click();
  await page.waitForFunction(() => !document.querySelector('#modal').open);
  await page.getByLabel('New list name').fill('Weekend');
  await page.getByRole('button',{name:'Create list',exact:true}).click();
  await page.getByRole('heading',{name:'Weekend',exact:true}).waitFor();
  await page.getByLabel('Followed account',{exact:true}).selectOption('sample@example.test');
  await page.getByRole('button',{name:'Add to list',exact:true}).click();
  await page.getByRole('button',{name:'Weekend 1 matched',exact:true}).waitFor();
  await page.getByRole('button',{name:'Remove',exact:true}).click();
  await page.getByRole('button',{name:'Weekend 0 matched',exact:true}).waitFor();
  await page.getByRole('button',{name:'Delete list',exact:true}).click();
  await page.getByRole('dialog').getByRole('button',{name:'Delete list',exact:true}).click();
  await page.waitForFunction(() => !document.querySelector('#modal').open);
  if (await page.getByRole('button',{name:/Weekend/}).count()) throw Error('Delete failed');
  await page.getByRole('button',{name:'Close circle 1 matched',exact:true}).click();
  await page.setViewportSize({width:1440,height:1000});
  await page.evaluate(() => window.scrollTo(0,0));
  await page.screenshot({path:'output/playwright/lists-desktop.png',fullPage:true});
  for (const width of [1440,1024,820,600,390,320]) {
    await page.setViewportSize({width,height:900});
    if (!await page.evaluate(() => document.documentElement.scrollWidth<=innerWidth)) throw Error('List overflow at '+width);
  }
  await page.setViewportSize({width:390,height:844});
  await page.screenshot({path:'output/playwright/lists-mobile.png',fullPage:true});
  await page.reload();
  await page.getByRole('button',{name:'Manage lists',exact:true}).click();
  await page.getByRole('button',{name:'Close circle 1 matched',exact:true}).click();
  await page.getByRole('button',{name:'Read this list',exact:true}).click();
  await page.waitForFunction(() => document.querySelectorAll('.board .post').length === 1);
  if (await page.locator('#feed-list').inputValue() !== id) throw Error('List identity changed');
  if (!await page.locator('.board .author-info').first().textContent().then(t => t.includes('Another author'))) throw Error('Explicit migration match failed');
  await page.locator('#feed-list').selectOption('');
  await page.waitForFunction(() => document.querySelectorAll('.board .post').length === 2);
  if (errors.length) throw Error(errors.join('; '));
  return {passed:true,importPreview:true,multiMembership:true,reimportPreservesEdits:true,manualMatching:true,createRenameDelete:true,feedAndPeopleFilters:true,persistedAcrossReload:true,responsiveWidths:[1440,1024,820,600,390,320],errors};
}
