async page => {
  page.setDefaultTimeout(10000);
  await page.context().setOffline(false);
  await page.unroute("**/api/**");
  const errors=[], writes=[];
  page.on('pageerror',error=>errors.push(error.message));
  const account={id:'20',acct:'person@example.social',display_name:'A public author'};
  const make=(id,extra={})=>({id:String(id),account,created_at:'2026-10-18T08:00:00Z',url:'https://example.social/posts/'+id,uri:'https://example.social/posts/'+id,visibility:'public',content:'<p>A story worth sharing.</p>',media_attachments:[],tags:[],favourited:false,bookmarked:false,reblogged:false,reblogs_count:2,...extra});
  const posts=[make(1),make(2,{visibility:'private'}),make(3,{visibility:'unlisted'}),make(4,{visibility:'direct'}),make(5,{visibility:undefined}),make(6,{account:{id:'30',acct:'https://rss.example/feed',display_name:'RSS source'}})];
  let failBoost=false, holdBoost=false, releaseBoost;
  const apiRoute=async route=>{
    const request=route.request(), path=new URL(request.url()).pathname;
    const respond=options=>route.fulfill({...options,headers:{"cache-control":"no-store"}});
    if(path==='/api/session')return respond({json:{phase:'full',hasAccount:true,csrf:'fixture',connection:{id:'10',acct:'reader',display_name:'Test reader'},preferences:{theme:'dark',font:'sans',size:18},passkeys:[],sessions:[],backupRemaining:8}});
    if(path==='/api/preferences')return respond({json:{ok:true}});
    if(path==='/api/wordpress/feed')return respond({json:{posts,next:null}});
    if(path==='/api/wordpress/post/1')return respond({json:posts[0]});
    if(path==='/api/wordpress/post/1/context')return respond({json:{ancestors:[],descendants:[]}});
    if(path.endsWith('/reblog')||path.endsWith('/unreblog')){
      writes.push(path);
      if(holdBoost)await new Promise(resolve=>releaseBoost=resolve);
      if(failBoost)return respond({status:502,json:{error:'WordPress could not confirm this boost.'}});
      posts[0].reblogged=path.endsWith('/reblog'); posts[0].reblogs_count=posts[0].reblogged?3:2;
      return respond({json:posts[0]});
    }
    return respond({status:404,json:{error:'Unknown fixture route'}});
  };
  await page.route('**/api/**',apiRoute);
  await page.setViewportSize({width:1440,height:1000});
  await page.goto('http://127.0.0.1:4323/');
  await page.locator('.post[data-id="1"]').waitFor();
  await page.evaluate(async()=>{await navigator.serviceWorker.ready;});
  await page.waitForFunction(()=>!!navigator.serviceWorker.controller);
  const cdp=await page.context().newCDPSession(page);
  const manifest=await cdp.send('Page.getAppManifest');
  if(manifest.errors?.length)throw Error('Manifest errors '+JSON.stringify(manifest.errors));
  const installability=await cdp.send('Page.getInstallabilityErrors');
  if(installability.installabilityErrors?.length)throw Error('Installability errors '+JSON.stringify(installability));
  const boost=page.locator('.post[data-id="1"] [data-do=boost]');
  for(const id of [2,3,4,5,6])if(!await page.locator(`.post[data-id="${id}"] [data-do=boost]`).isDisabled())throw Error('Unsafe boost available '+id);
  failBoost=true;await boost.click();await page.locator('#status-message').filter({hasText:'WordPress could not confirm this boost.'}).waitFor();
  if(await boost.getAttribute('aria-pressed')!=='false')throw Error('Optimistic boost after failure');
  failBoost=false;holdBoost=true;await boost.click();
  await page.waitForFunction(()=>document.querySelector('.post[data-id="1"] [data-do=boost]').disabled);
  if(writes.length!==2)throw Error('Duplicate boost');holdBoost=false;releaseBoost();
  await page.waitForFunction(()=>document.querySelector('.post[data-id="1"] [data-do=boost]').getAttribute('aria-pressed')==='true');
  if(await boost.getAttribute('aria-label')!=='Undo boost post')throw Error('Boost label');
  await page.locator('.post[data-id="1"] .read-button').click();
  await page.locator('.reader [data-do=boost]').waitFor();
  if(await page.locator('.reader [data-do=boost]').getAttribute('aria-pressed')!=='true')throw Error('Reader boost state');
  await page.locator('.reader [data-do=boost]').click();
  await page.waitForFunction(()=>document.querySelector('.reader [data-do=boost]').getAttribute('aria-pressed')==='false');
  if(await boost.getAttribute('aria-pressed')!=='false')throw Error('Card did not reflect undo');
  await page.locator('.reader-back').click();
  await page.getByRole('button',{name:'Account',exact:true}).first().click();
  await page.getByRole('heading',{name:'Install Asterism',exact:true}).waitFor();
  await page.evaluate(()=>{const e=new Event('beforeinstallprompt',{cancelable:true});Object.defineProperty(e,'prompt',{value:async()=>{window.promptTest=(window.promptTest||0)+1;return {outcome:'dismissed'};}});window.dispatchEvent(e);});
  await page.getByRole('button',{name:'Install Asterism',exact:true}).click();
  if(await page.evaluate(()=>window.promptTest)!==1)throw Error('Install button did not invoke prompt');
  await page.getByRole('button',{name:'Installation instructions',exact:true}).click();
  if(!await page.locator('.install-help').evaluate(e=>e.open))throw Error('Manual installation instructions');
  for(const width of [1440,820,390,320]){
    await page.setViewportSize({width,height:900});
    if(await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth))throw Error('Install layout overflow '+width);
  }
  await page.setViewportSize({width:1440,height:1000});
  await page.getByRole('button',{name:'Feed',exact:true}).first().click();
  await page.locator('.post[data-id="1"] .read-button').click();
  await page.locator('#reply-text').fill('An unfinished reply that must survive the update prompt');
  await page.locator('.reader-back').click();
  await page.evaluate(async()=>{await fetch('/__test/update-worker',{method:'POST'});const r=await navigator.serviceWorker.getRegistration();await r.update();});
  await page.getByRole('button',{name:'Update app',exact:true}).waitFor();
  await page.getByRole('button',{name:'Update app',exact:true}).click();
  await page.getByText('Finish or clear your reply drafts and close other dialogs before updating.',{exact:true}).waitFor();
  if(!await page.evaluate(async()=>!!(await navigator.serviceWorker.getRegistration()).waiting))throw Error('Update activated over draft');
  await page.locator('.post[data-id="1"] .read-button').click();
  if(await page.locator('#reply-text').inputValue()!=='An unfinished reply that must survive the update prompt')throw Error('Draft lost');
  await page.locator('#reply-text').fill('');
  await page.locator('.reader-back').click();
  await Promise.all([page.waitForEvent('load'),page.getByRole('button',{name:'Update app',exact:true}).click()]);
  await page.locator('.post[data-id="1"]').waitFor();
  if(await page.evaluate(async()=>!!(await navigator.serviceWorker.getRegistration()).waiting))throw Error('Update did not activate');
  const cached=await page.evaluate(async()=>{const entries=[];for(const key of await caches.keys()){const c=await caches.open(key);entries.push(...(await c.keys()).map(r=>new URL(r.url).pathname));}return entries;});
  if(cached.some(path=>path.startsWith('/api/')||path==='/'||path==='/index.html'))throw Error('Private or dynamic response cached '+JSON.stringify(cached));
  await page.unroute('**/api/**');
  await page.context().setOffline(true);
  const apiOffline=await page.evaluate(async()=>{try{await fetch('/api/session',{cache:'no-store'});return false;}catch{return true;}});
  if(!apiOffline)throw Error('API served offline');
  await page.goto('http://127.0.0.1:4323/');
  await page.getByRole('heading',{name:'A moment to reconnect.',exact:true}).waitFor();
  if((await page.locator('body').innerText()).includes('A story worth sharing'))throw Error('Private feed leaked offline');
  await page.screenshot({path:'output/playwright/asterism-offline.png'});
  await page.context().setOffline(false);
  await page.route('**/api/**',apiRoute);
  await page.goto('http://127.0.0.1:4323/');
  await page.locator('.post[data-id="1"]').waitFor();
  await page.screenshot({path:'output/playwright/asterism-boost-desktop.png'});
  await page.setViewportSize({width:390,height:844});
  await page.screenshot({path:'output/playwright/asterism-boost-mobile.png'});
  await page.getByRole('button',{name:'Account',exact:true}).last().click();
  await page.locator('[data-install-card]').waitFor();
  await page.locator('[data-install-card]').scrollIntoViewIfNeeded();
  await page.screenshot({path:'output/playwright/asterism-install-mobile.png'});
  await page.evaluate(()=>window.dispatchEvent(new Event('appinstalled')));
  if(!await page.getByRole('button',{name:'Installed',exact:true}).isDisabled())throw Error('Installed status was not retained');
  if(errors.length)throw Error('Browser errors '+errors.join('; '));
  return {passed:true,boostWrites:writes,installability,cachedPublicFiles:cached.length,draftProtected:true,offlineApiBlocked:true};
}
